    ```
    from dataclasses import dataclass, field
    from datetime import datetime, timezone
    from enum import Enum

    class ComponentHealth(str, Enum):
        HEALTHY = "healthy"
        DEGRADED = "degraded"
        UNHEALTHY = "unhealthy"

    @dataclass
    class HealthStatus:
        ib_connected: bool
        ib_reconnect_count: int
        db_connected: bool
        redis_connected: bool
        trading_mode: str
        uptime_seconds: float
        last_check: datetime
        overall: ComponentHealth  # Computed from components

    class HealthMonitor:
        def __init__(self, connection_manager: IBConnectionManager,
                     db_engine: AsyncEngine,
                     redis_client: Redis,
                     settings: Settings):
            self.connection_manager = connection_manager
            self.db_engine = db_engine
            self.redis_client = redis_client
            self.settings = settings
            self._start_time = datetime.now(timezone.utc)
            self._logger = structlog.get_logger(component="health_monitor")

        async def check_health(self) -> HealthStatus:
            # 1. Check IB: connection_manager.is_connected
            # 2. Check DB: try a simple query (SELECT 1) with timeout
            # 3. Check Redis: try PING with timeout
            # 4. Compute overall:
            #    - All healthy -> HEALTHY
            #    - IB disconnected but DB+Redis ok -> DEGRADED
            #    - Any critical component down -> UNHEALTHY
            # 5. Return HealthStatus dataclass
            # 6. Log health status at appropriate level

        async def _check_db(self) -> bool:
            try:
                async with self.db_engine.connect() as conn:
                    await conn.execute(text("SELECT 1"))
                return True
            except Exception:
                return False

        async def _check_redis(self) -> bool:
            try:
                return await self.redis_client.ping()
            except Exception:
                return False

        def get_uptime(self) -> float:
            return (datetime.now(timezone.utc) - self._start_time).total_seconds()
    ```

    ```
    class TradingApp:
        def __init__(self, settings: Settings):
            self.settings = settings
            self.connection_manager: IBConnectionManager | None = None
            self.db_engine: AsyncEngine | None = None
            self.session_factory = None
            self.redis_client: Redis | None = None
            self.health_monitor: HealthMonitor | None = None
            self.kill_switch: KillSwitch | None = None
            self.order_tracker: OrderTracker | None = None
            self._logger = structlog.get_logger(component="app")

        async def startup(self):
            # 1. Configure structlog (keep existing logic)
            # 2. Log startup banner with trading mode (keep existing logic)
            # 3. Create DB engine and session factory
            self.db_engine = create_db_engine(self.settings.database.url,
                                               pool_size=self.settings.database.pool_size)
            self.session_factory = create_session_factory(self.db_engine)
            # 4. Create Redis client
            self.redis_client = create_redis_client(self.settings.redis.url)
            # 5. Create IB connection manager (but don't connect yet)
            self.connection_manager = IBConnectionManager(self.settings)
            # 6. Create order tracker
            self.order_tracker = OrderTracker(self.session_factory)
            # 7. Create kill switch
            self.kill_switch = KillSwitch(self.connection_manager.ib)
            # 8. Create health monitor
            self.health_monitor = HealthMonitor(
                self.connection_manager, self.db_engine,
                self.redis_client, self.settings
            )
            # 9. Log "startup.complete" with component list

        async def connect_ib(self):
            # Separate from startup -- allows running without IB for testing
            if self.connection_manager:
                await self.connection_manager.connect()

        async def shutdown(self):
            # 1. Disconnect IB (if connected)
            if self.connection_manager:
                await self.connection_manager.disconnect()
            # 2. Close Redis
            if self.redis_client:
                await self.redis_client.aclose()
            # 3. Dispose DB engine
            if self.db_engine:
                await self.db_engine.dispose()
            # 4. Log "shutdown.complete"

    async def main():
        settings = Settings()  # Or Settings.load() if custom loading
        app = TradingApp(settings)
        await app.startup()

        # Don't auto-connect to IB in main() -- let caller decide
        # This supports running just infrastructure (DB, Redis) without IB

        try:
            # Keep running until interrupted
            while True:
                await asyncio.sleep(1)
        except (KeyboardInterrupt, asyncio.CancelledError):
            pass
        finally:
            await app.shutdown()

    if __name__ == "__main__":
        asyncio.run(main())
    ```
